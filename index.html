async function loadData() { try { // Ambil file data.json yang dibuat oleh GitHub Actions const response = await fetch('./data.json'); const result = await response.json(); // Tampilkan data ke HTML document.getElementById('data-container').innerText = JSON.stringify(result, null, 2); document.getElementById('loading').style.display = 'none'; } catch (error) { document.getElementById('loading').innerText = 'Data belum tersedia.'; } } loadData();
